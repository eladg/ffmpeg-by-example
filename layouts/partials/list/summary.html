{{- /* An example's one-line summary: the first sentence of its description as plain text, at most 200 characters. Used by the homepage list and llms.txt. */ -}} {{- /* code spans are escaped twice, so unescape twice */ -}} {{- $text := .Description | markdownify | plainify | htmlUnescape | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- $summary := $text -}} {{- with findRE `^.+?[.!?](\s|$)` $text 1 }}{{ $summary = index . 0 | strings.TrimSpace }}{{ end -}} {{- if gt (len $summary) 200 }}{{ $summary = printf "%s…" (substr $summary 0 199 | strings.TrimSpace) }}{{ end -}} {{- return $summary -}}
